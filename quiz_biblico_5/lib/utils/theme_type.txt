import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

enum AppTheme {
  system,
  light,
  dark,
  sepia,
  highContrast,
  orangeLight,
}

class ThemeProvider extends ChangeNotifier {
  ThemeData _themeData;
  AppTheme _selectedTheme;

  ThemeProvider()
      : _selectedTheme = AppTheme.dark,
        _themeData = _buildDarkTheme();

  ThemeData get themeData => _themeData;
  AppTheme get selectedTheme => _selectedTheme;

  void setTheme(AppTheme theme) {
    _selectedTheme = theme;
    switch (theme) {
      case AppTheme.dark:
        _themeData = _buildDarkTheme();
        break;
      case AppTheme.light:
        _themeData = _buildLightTheme();
        break;
      case AppTheme.sepia:
        _themeData = _buildSepiaTheme();
        break;
      case AppTheme.highContrast:
        _themeData = _buildHighContrastTheme();
        break;
      case AppTheme.orangeLight:
        _themeData = _buildOrangeLightTheme();
        break;
      case AppTheme.system:
        final brightness = WidgetsBinding.instance.platformDispatcher.platformBrightness;
        _themeData = brightness == Brightness.dark ? _buildDarkTheme() : _buildLightTheme();
        break;
    }
    notifyListeners();
  }

  static ThemeData _buildLightTheme() {
    return ThemeData(
      brightness: Brightness.light,
      primaryColor: Colors.blue,
      scaffoldBackgroundColor: Colors.white,
      appBarTheme: const AppBarTheme(
        backgroundColor: Colors.white,
        titleTextStyle: TextStyle(color: Colors.black, fontSize: 30, fontFamily: 'Trocchi'),
        iconTheme: IconThemeData(color: Colors.black),
      ),
      iconTheme: const IconThemeData(color: Colors.black),
      textTheme: const TextTheme(
        headlineMedium: TextStyle(color: Colors.black, fontSize: 23, fontFamily: 'Trocchi'),
        bodyMedium: TextStyle(color: Colors.black, fontSize: 18, fontFamily: 'Trocchi'),
        titleMedium: TextStyle(color: Colors.black, fontSize: 18, fontFamily: 'Trocchi'),
        titleSmall: TextStyle(color: Colors.black54, fontSize: 15, fontFamily: 'Trocchi'),
      ),
    );
  }

  static ThemeData _buildDarkTheme() {
    return ThemeData(
      brightness: Brightness.dark,
      primaryColor: Colors.white24,
      secondaryHeaderColor: Colors.black45,
      iconTheme: const IconThemeData(color: Colors.white),
      appBarTheme: const AppBarTheme(
        titleTextStyle: TextStyle(fontSize: 30, color: Colors.white, fontFamily: 'Trocchi'),
      ),
      textTheme: const TextTheme(
        headlineMedium: TextStyle(fontSize: 23, color: Colors.white, fontFamily: 'Trocchi'),
        bodyMedium: TextStyle(color: Colors.white, fontSize: 18, fontFamily: 'Trocchi'),
        titleMedium: TextStyle(color: Colors.white, fontSize: 18.0, fontFamily: 'Trocchi'),
        titleSmall: TextStyle(color: Colors.black, fontSize: 15.0, fontFamily: 'Trocchi'),
      ),
    );
  }

  static ThemeData _buildSepiaTheme() {
    return ThemeData(
      brightness: Brightness.light,
      scaffoldBackgroundColor: const Color(0xFFF5EAD6),
      primaryColor: const Color(0xFFC8B08E),
      secondaryHeaderColor: const Color(0xFFA38D6D),
      appBarTheme: const AppBarTheme(
        backgroundColor: Color(0xFFD4C2A2),
        titleTextStyle: TextStyle(fontSize: 30, color: Colors.brown, fontFamily: 'Trocchi'),
      ),
      iconTheme: const IconThemeData(color: Colors.brown),
      textTheme: const TextTheme(
        headlineMedium: TextStyle(fontSize: 23, color: Colors.brown, fontFamily: 'Trocchi'),
        bodyMedium: TextStyle(fontSize: 18, color: Colors.brown, fontFamily: 'Trocchi'),
        titleMedium: TextStyle(fontSize: 18, color: Colors.brown, fontFamily: 'Trocchi'),
        titleSmall: TextStyle(fontSize: 15, color: Colors.black, fontFamily: 'Trocchi'),
      ),
    );
  }

  static ThemeData _buildHighContrastTheme() {
    return ThemeData(
      brightness: Brightness.dark,
      scaffoldBackgroundColor: Colors.black,
      primaryColor: Colors.white,
      secondaryHeaderColor: Colors.white70,
      iconTheme: const IconThemeData(color: Colors.white),
      appBarTheme: const AppBarTheme(
        backgroundColor: Colors.black,
        titleTextStyle: TextStyle(fontSize: 30, color: Colors.white, fontFamily: 'Trocchi'),
      ),
      textTheme: const TextTheme(
        headlineMedium: TextStyle(fontSize: 23, color: Colors.white, fontFamily: 'Trocchi'),
        bodyMedium: TextStyle(fontSize: 18, color: Colors.white, fontFamily: 'Trocchi'),
        titleMedium: TextStyle(fontSize: 18, color: Colors.white, fontFamily: 'Trocchi'),
        titleSmall: TextStyle(fontSize: 15, color: Colors.yellow, fontFamily: 'Trocchi'),
      ),
      colorScheme: const ColorScheme.highContrastDark(),
    );
  }

  static ThemeData _buildOrangeLightTheme() {
    return ThemeData(
      brightness: Brightness.light,
      scaffoldBackgroundColor: const Color(0xFFFFF3E0),
      primaryColor: const Color(0xFFFF9800),
      secondaryHeaderColor: const Color(0xFFFFCC80),
      iconTheme: const IconThemeData(color: Color(0xFFEF6C00)),
      appBarTheme: const AppBarTheme(
        backgroundColor: Color(0xFFFFA726),
        titleTextStyle: TextStyle(fontSize: 30, color: Colors.white, fontFamily: 'Trocchi'),
      ),
      textTheme: const TextTheme(
        headlineMedium: TextStyle(fontSize: 23, color: Colors.deepOrange, fontFamily: 'Trocchi'),
        bodyMedium: TextStyle(fontSize: 18, color: Colors.black87, fontFamily: 'Trocchi'),
        titleMedium: TextStyle(fontSize: 18, color: Colors.black87, fontFamily: 'Trocchi'),
        titleSmall: TextStyle(fontSize: 15, color: Colors.black54, fontFamily: 'Trocchi'),
      ),
    );
  }
}

class ThemeSelectionDialog extends StatelessWidget {
  const ThemeSelectionDialog({super.key});

  @override
  Widget build(BuildContext context) {
    final provider = Provider.of<ThemeProvider>(context);
    return AlertDialog(
      title: const Text('Escolha o Tema'),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        children: AppTheme.values.map((theme) {
          final isSelected = provider.selectedTheme == theme;
          return ListTile(
            leading: Icon(
              isSelected ? Icons.check_circle : Icons.circle_outlined,
              color: isSelected ? Colors.green : null,
            ),
            title: Text(_getThemeName(theme)),
            onTap: () {
              provider.setTheme(theme);
              Navigator.of(context).pop();
            },
          );
        }).toList(),
      ),
    );
  }

  String _getThemeName(AppTheme theme) {
    switch (theme) {
      case AppTheme.system:
        return 'Sistema';
      case AppTheme.light:
        return 'Claro';
      case AppTheme.dark:
        return 'Escuro';
      case AppTheme.sepia:
        return 'Sépia';
      case AppTheme.highContrast:
        return 'Alto Contraste';
      case AppTheme.orangeLight:
        return 'Luz Laranja';
    }
  }
}
